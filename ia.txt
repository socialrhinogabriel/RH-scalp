// Clamp value între min și max
f_clamp(float _val, float _min, float _max) =>
    math.max(_min, math.min(_max, _val))

// Timeframe to minutes
f_tfToMinutes() =>
    float _mult = timeframe.multiplier
    timeframe.isseconds ? _mult / 60.0 :
     timeframe.isminutes ? _mult :
     timeframe.isdaily ? _mult * 1440.0 :
     timeframe.isweekly ? _mult * 10080.0 :
     timeframe.ismonthly ? _mult * 43200.0 :
     _mult

// Format timeframe pentru display
f_formatTF(string _tf) =>
    _tf == "1" ? "1m" :
     _tf == "3" ? "3m" :
     _tf == "5" ? "5m" :
     _tf == "15" ? "15m" :
     _tf == "30" ? "30m" :
     _tf == "60" ? "1H" :
     _tf == "120" ? "2H" :
     _tf == "240" ? "4H" :
     _tf == "D" ? "1D" :
     _tf == "W" ? "1W" :
     _tf

// Format price
f_formatPrice(float _price) =>
    _price >= 10000 ? str.tostring(_price, "#") :
     _price >= 1000 ? str.tostring(_price, "#.#") :
     _price >= 100 ? str.tostring(_price, "#.##") :
     _price >= 1 ? str.tostring(_price, "#.###") :
     str.tostring(_price, "#.#####")

// Hyperbolic tangent
f_tanh(float _x) =>
    float _exp2x = math.exp(2 * _x)
    (_exp2x - 1) / (_exp2x + 1)

// Dual Pole Filter (Super Smoother)
f_dualPoleFilter(float _src, float _len) =>
    float _a = math.exp(-1.414 * math.pi / _len)
    float _b = 2 * _a * math.cos(1.414 * math.pi / _len)
    float _c2 = _b
    float _c3 = -_a * _a
    float _c1 = 1 - _c2 - _c3
    var float _filt = na
    _filt := _c1 * (_src + nz(_src[1])) / 2 + _c2 * nz(_filt[1]) + _c3 * nz(_filt[2])
    _filt

// Normalize Derivative
f_normalizeDeriv(float _src, int _len) =>
    float _ema = ta.ema(_src, _len)
    float _deriv = _src - _src[1]
    float _norm = ta.stdev(_deriv, _len)
    _norm > 0 ? _deriv / _norm : 0

// Kernel Gaussian
f_kernelGaussian(float _src, int _lookback, float _bandwidth) =>
    float _sum = 0.0
    float _weightSum = 0.0
    for i = 0 to _lookback - 1
        float _weight = math.exp(-math.pow(i, 2) / (2 * math.pow(_bandwidth, 2)))
        _sum += _src[i] * _weight
        _weightSum += _weight
    _weightSum > 0 ? _sum / _weightSum : _src

// Moving Average universal
f_ma(float _src, int _len, string _type) =>
    switch _type
        "SMA" => ta.sma(_src, _len)
        "EMA" => ta.ema(_src, _len)
        "WMA" => ta.wma(_src, _len)
        "RMA" => ta.rma(_src, _len)
        "HMA" => ta.hma(_src, _len)
        "VWMA" => ta.vwma(_src, _len)
        => ta.ema(_src, _len)

// Volume Ratio
f_volumeRatio(int _len) =>
    float _avgVol = ta.sma(volume, _len)
    _avgVol > 0 ? volume / _avgVol : 1.0

// Choppiness Index
f_choppiness(int _len) =>
    float _atr = ta.atr(1)
    float _sumATR = math.sum(_atr, _len)
    float _high = ta.highest(high, _len)
    float _low = ta.lowest(low, _len)
    float _range = _high - _low
    _range > 0 ? 100 * math.log10(_sumATR / _range) / math.log10(_len) : 50

// Check Fibonacci Ratio
f_checkFibRatio(float _move1, float _move2, float _targetRatio, float _tolerance) =>
    float _actualRatio = _move1 != 0 ? math.abs(_move2 / _move1) : 0
    math.abs(_actualRatio - _targetRatio) <= _tolerance

// Probability to Grade
f_probToGrade(float _prob) =>

     _prob >= 0.55 ? "B" :
     _prob >= 0.45 ? "C" :

f_colorGradient(float _val, float _min, float _max, color _colMin, color _colMax) =>

    color.from_gradient(_pct, 0, 1, _colMin, _colMax)

// Componentele lumânării
f_candleComponents() =>
    float _body = math.abs(close - open)
    float _upperWick = high - math.max(close, open)
    float _lowerWick = math.min(close, open) - low
    float _fullRange = high - low
    float _bodyPct = _fullRange > 0 ? _body / _fullRange : 0
    bool _isBullish = close > open
    bool _isBearish = close < open
    bool _isDoji = _bodyPct < 0.1
    [_body, _upperWick, _lowerWick, _fullRange, _bodyPct, _isBullish, _isBearish, _isDoji]

// Detectare Pin Bar
f_isPinBar() =>
    [_body, _upperWick, _lowerWick, _fullRange, _bodyPct, _isBullish, _isBearish, _isDoji] = f_candleComponents()
    bool _bullishPin = _lowerWick > _body * 2 and _upperWick < _body * 0.5 and _bodyPct < 0.4
    bool _bearishPin = _upperWick > _body * 2 and _lowerWick < _body * 0.5 and _bodyPct < 0.4
    [_bullishPin, _bearishPin]

// Detectare Engulfing
f_isEngulfing() =>
    float _body = math.abs(close - open)
    float _prevBody = math.abs(close[1] - open[1])
    bool _bullishEngulf = close > open and close[1] < open[1] and close > open[1] and open < close[1] and _body > _prevBody
    bool _bearishEngulf = close < open and close[1] > open[1] and close < open[1] and open > close[1] and _body > _prevBody
    [_bullishEngulf, _bearishEngulf]

// Detectare Hammer/Hanging Man
f_isHammer() =>
    [_body, _upperWick, _lowerWick, _fullRange, _bodyPct, _isBullish, _isBearish, _isDoji] = f_candleComponents()
    float _avgBody = ta.sma(math.abs(close - open), 20)
    bool _isHammer = _lowerWick > _body * 2 and _upperWick < _body * 0.3 and _body > _avgBody * 0.5
    bool _isInvertedHammer = _upperWick > _body * 2 and _lowerWick < _body * 0.3 and _body > _avgBody * 0.5
    [_isHammer, _isInvertedHammer]

// Detectare Inside/Outside Bar

// SMART TRAIL - ATR Trailing Stop inteligent
// Original: LAF.getSmartTrail(10, 6, 8)
// Returnează: [trailLine, fillerLine, direction]
// ═══════════════════════════════════════════════════════════════════════════════

f_getSmartTrail(int _length, int _atrPeriod, float _mult) =>
    float _atr = ta.atr(_atrPeriod)
    float _src = close
    
    // Calculăm benzile
    float _upperBand = _src + _atr * _mult
    float _lowerBand = _src - _atr * _mult
    
    // Variabile persistente
    var float _trail = na
    var float _filler = na
    var string _dir = "neutral"
    var int _trend = 0
    
    // Determinăm trendul
    float _prevTrail = nz(_trail[1], _src)
    
    if _src > _prevTrail
        _trail := math.max(_prevTrail, _lowerBand)
        _filler := _trail - _atr * 0.5
        _dir := "long"
        _trend := 1
    else
        _trail := math.min(_prevTrail, _upperBand)
        _filler := _trail + _atr * 0.5
        _dir := "short"
        _trend := -1
    
    // Smoothing pentru aspect vizual mai curat
    float _smoothTrail = ta.ema(_trail, 3)
    float _smoothFiller = ta.ema(_filler, 3)
    
    [_smoothTrail, _smoothFiller, _dir, _trend]

// TREND CATCHER - Supertrend modificat
// Original: LAF.getTrendCatcher()
// Returnează: [line, color]
// ═══════════════════════════════════════════════════════════════════════════════

f_getTrendCatcher(int _length, float _mult) =>
    float _atr = ta.atr(_length)
    float _src = hl2
    
    float _upperBand = _src + _mult * _atr
    float _lowerBand = _src - _mult * _atr
    
    var float _supertrend = na
    var int _direction = 1
    
    float _prevUpperBand = nz(_upperBand[1], _upperBand)
    float _prevLowerBand = nz(_lowerBand[1], _lowerBand)
    float _prevSupertrend = nz(_supertrend[1], _src)
    
    // Ajustare benzi
    _lowerBand := _lowerBand > _prevLowerBand or close[1] < _prevLowerBand ? _lowerBand : _prevLowerBand
    _upperBand := _upperBand < _prevUpperBand or close[1] > _prevUpperBand ? _upperBand : _prevUpperBand
    
    // Determinare direcție
    if _prevSupertrend == _prevUpperBand
        _direction := close > _upperBand ? 1 : -1
    else
        _direction := close < _lowerBand ? -1 : 1
    
    _supertrend := _direction == 1 ? _lowerBand : _upperBand
    
    // Culoare bazată pe direcție
    color _color = _direction == 1 ? C_BULL : C_BEAR
    
    [_supertrend, _color, _direction]

// TREND TRACER - Hull Moving Average adaptiv
// Original: LAF.getTrendTracer()
// Returnează: [line, direction_color]
// ═══════════════════════════════════════════════════════════════════════════════

f_getTrendTracer(int _length, int _smooth) =>
    // Calculăm HMA modificat
    int _halfLen = math.max(1, _length / 2)
    int _sqrtLen = math.max(1, math.round(math.sqrt(_length)))
    
    float _wma1 = ta.wma(close, _halfLen)
    float _wma2 = ta.wma(close, _length)
    float _hma = ta.wma(2 * _wma1 - _wma2, _sqrtLen)
    
    // Smoothing suplimentar
    float _tracer = ta.ema(_hma, _smooth)
    
    // Determinare direcție și viteză
    bool _rising = _tracer > _tracer[1]
    bool _falling = _tracer < _tracer[1]
    bool _bullish = _rising and close > _tracer
    bool _bearish = _falling and close < _tracer
    
    // Determinare culoare
    color _color = _bullish ? C_BULL : _bearish ? C_BEAR : C_NEUTRAL
    
    int _dir = _bullish ? 1 : _bearish ? -1 : 0
    
    [_tracer, _color, _dir]

f_getTrendStrengthMetric(int _adxLen, string _adxSmooth, int _diLen, string _diSmooth) =>
    [_plusDI, _minusDI, _adx] = ta.dmi(_adxLen, _diLen)
    
    // Aplicăm smoothing pe ADX
    float _smoothADX = switch _adxSmooth
        "SMA" => ta.sma(_adx, 5)
        "EMA" => ta.ema(_adx, 5)
        "RMA" => ta.rma(_adx, 5)
        "WMA" => ta.wma(_adx, 5)
        =>      _adx
    
    // Calculăm diferența DI
    float _diDiff = _plusDI - _minusDI
    
    // Calculăm trend strength cu semn (pozitiv = bullish, negativ = bearish)
    float _strength = _smoothADX * (_diDiff > 0 ? 1 : -1)
    
    // Normalizăm la -100 / +100
    float _normalized = f_clamp(_strength, -100, 100)
    
    _normalized

// SQUEEZE METRIC - Bollinger Bands vs Keltner Channel
// Original: LAF.getSqueezeMetric(45, 20)
// Returnează: float (0-100, >80 = squeeze puternic)
// ═══════════════════════════════════════════════════════════════════════════════

f_getSqueezeMetric(int _bbLen, int _kcLen) =>
    // Bollinger Bands
    [_bbMid, _bbUp, _bbLow] = ta.bb(close, _bbLen, 2.0)
    float _bbWidth = (_bbUp - _bbLow) / _bbMid * 100
    
    // Keltner Channel
    float _kcMid = ta.ema(close, _kcLen)
    float _kcRange = ta.atr(_kcLen) * 1.5
    float _kcUp = _kcMid + _kcRange
    float _kcLow = _kcMid - _kcRange
    float _kcWidth = (_kcUp - _kcLow) / _kcMid * 100
    
    // Squeeze: când BB este în interiorul KC
    bool _inSqueeze = _bbUp < _kcUp and _bbLow > _kcLow
    
    // Metric: cât de strâns este squeeze-ul
    float _ratio = _kcWidth > 0 ? _bbWidth / _kcWidth * 100 : 50
    float _metric = _inSqueeze ? 100 - _ratio : _ratio
    
    f_clamp(_metric, 0, 100)

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE HELPER PENTRU NEO CLOUD (TREBUIE DEFINITĂ SEPARAT - NU NESTED)
// ═══════════════════════════════════════════════════════════════════════════════

f_avgWithATR(float _src, int _len, float _mult) =>
    float _atr = ta.atr(_len) * _mult
    float _up = hl2 + _atr
    float _dn = hl2 - _atr
    
    var float _upper = _up
    var float _lower = _dn
    
    _upper := _src[1] < _upper[1] ? math.min(_up, _upper[1]) : _up
    _lower := _src[1] > _lower[1] ? math.max(_dn, _lower[1]) : _dn
    
    var int _os = 0
    var float _max = high
    var float _min = low
    
    _os := _src > _upper ? 1 : _src < _lower ? 0 : _os[1]
    float _spt = _os == 1 ? _lower : _upper
    _max := ta.cross(_src, _spt) ? math.max(_src, _max[1]) : _os == 1 ? math.max(_src, _max[1]) : _spt
    _min := ta.cross(_src, _spt) ? math.min(_src, _min[1]) : _os == 0 ? math.min(_src, _min[1]) : _spt
    
    math.avg(_max, _min)

    float _kijun = f_avgWithATR(close, _kijunLen, 7)

    float _senkouB = f_avgWithATR(close, _spanLen, 15)
    
    // Determinăm cloud bullish/bearish
    bool _cloudBull = _senkouA > _senkouB
    bool _cloudBear = _senkouA < _senkouB
    
    // Crossovers
    bool _tkCrossUp = ta.crossover(_tenkan, _kijun)
    bool _tkCrossDown = ta.crossunder(_tenkan, _kijun)
    
    [_tenkan, _kijun, _senkouA, _senkouB, _cloudBull, _cloudBear, _tkCrossUp, _tkCrossDown]

// REVERSAL ZONES - Mean Reversion Channel
// ═══════════════════════════════════════════════════════════════════════════════

f_getReversalZones(float _src, int _length, float _innerMult, float _outerMult) =>
    // SuperSmoother pentru mean line
    float _mean = f_dualPoleFilter(_src, _length)
    
    // Range bazat pe TR smoothed
    float _tr = ta.tr(true)
    float _range = f_dualPoleFilter(_tr, _length)
    
    // Multiplicatori (folosim Pi pentru simetrie)
    float _pi = 3.14159265359
    float _innerMultPi = _pi * _innerMult
    float _outerMultPi = _pi * _outerMult
    
    // Benzi interioare
    float _upInner = _mean + _range * _innerMultPi
    float _dnInner = _mean - _range * _innerMultPi
    
    // Benzi exterioare
    float _upOuter = _mean + _range * _outerMultPi
    float _dnOuter = _mean - _range * _outerMultPi
    
    // Condiții de overbought/oversold
    int _condition = 0
    
    if close > _mean
        if high >= _upOuter
            _condition := 3  // Strong overbought
        else if high >= _upInner
            _condition := 2  // Overbought
        else if high >= _mean + _range * _pi * 0.5
            _condition := 1  // Weak overbought
    else
        if low <= _dnOuter
            _condition := -3  // Strong oversold
        else if low <= _dnInner
            _condition := -2  // Oversold
        else if low <= _mean - _range * _pi * 0.5
            _condition := -1  // Weak oversold
    
    [_mean, _upInner, _dnInner, _upOuter, _dnOuter, _condition]

f_supertrend(float _mult, int _len) =>
    float _atr = ta.atr(_len)
    float _up = hl2 + _mult * _atr
    float _dn = hl2 - _mult * _atr
    
    var float _upperBand = na
    var float _lowerBand = na

// WAVETREND CLASSIC (Original)
// ═══════════════════════════════════════════════════════════════════════════════

f_waveTrendClassic(float _src, int _channelLen, int _avgLen, int _maLen) =>
    // Channel calculation
    float _esa = ta.ema(_src, _channelLen)
    float _de = ta.ema(math.abs(_src - _esa), _channelLen)
    float _ci = _de != 0 ? (_src - _esa) / (0.015 * _de) : 0
    
    // Wave trend lines
    float _wt1 = ta.ema(_ci, _avgLen)
    float _wt2 = ta.sma(_wt1, _maLen)
    
    // Histogram
    float _hist = _wt1 - _wt2
    
    [_wt1, _wt2, _hist]

// WAVETREND 3D - Trei viteze pentru analiză multi-nivel
// ═══════════════════════════════════════════════════════════════════════════════

f_waveTrend3D(float _src, int _lookback, int _qmLen, float _speedMult) =>
    // Normalize derivative pentru normalizare
    float _nDeriv = f_normalizeDeriv(_src, _qmLen)
    
    // Hyperbolic tangent pentru limitare la [-1, 1]
    float _tanh = f_tanh(_nDeriv)
    
    // Dual pole filter pentru smoothing
    float _smoothed = f_dualPoleFilter(_tanh, _lookback * _speedMult)
    
    // Scalare bazată pe viteză
    float _result = _smoothed * _speedMult
    
    _result

// PARTEA 6/15: K-MEANS SUPERTREND

// Supertrend optimizat automat cu K-Means clustering
// Selectează cel mai performant factor ATR din multiple opțiuni
// ═══════════════════════════════════════════════════════════════════════════════

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Calculează Supertrend individual cu tracking performanță
// ═══════════════════════════════════════════════════════════════════════════════

f_calcSupertrendWithPerf(float _factor, int _atrLen, int _perfAlpha) =>
    float _atr = ta.atr(_atrLen)
    float _up = hl2 + _atr * _factor
    float _dn = hl2 - _atr * _factor
    
    var float _upper = na
    var float _lower = na
    var int _trend = 1
    var float _output = na
    var float _perf = 0.0
    
    // Inițializare
    if na(_upper)
        _upper := _up
        _lower := _dn
        _output := _dn
    
    // Update benzi
    _upper := close[1] < _upper[1] ? math.min(_up, _upper[1]) : _up
    _lower := close[1] > _lower[1] ? math.max(_dn, _lower[1]) : _dn
    
    // Determinare trend
    _trend := close > _upper[1] ? 1 : close < _lower[1] ? -1 : _trend[1]
    _output := _trend == 1 ? _lower : _upper
    
    // Performance tracking (exponential moving)
    float _diff = nz(math.sign(close[1] - _output[1]), 0)
    float _pnl = nz(close - close[1], 0) * _diff
    _perf := _perf + 2.0 / (_perfAlpha + 1) * (_pnl - _perf)
    
    [_output, _trend, _perf, _upper, _lower]

// ═══════════════════════════════════════════════════════════════════════════════
// INIȚIALIZARE ARRAYS PENTRU K-MEANS
// ═══════════════════════════════════════════════════════════════════════════════

// Calculăm numărul de factori
int km_numFactors = math.floor((i_kmeansMaxMult - i_kmeansMinMult) / i_kmeansStep) + 1
km_numFactors := math.min(km_numFactors, 20) // Limită pentru performanță

// Arrays pentru stocarea rezultatelor
var array<float> km_factors = array.new<float>(km_numFactors, 0.0)
var array<float> km_outputs = array.new<float>(km_numFactors, 0.0)
var array<float> km_perfs = array.new<float>(km_numFactors, 0.0)
var array<int> km_trends = array.new<int>(km_numFactors, 0)
var array<float> km_uppers = array.new<float>(km_numFactors, 0.0)
var array<float> km_lowers = array.new<float>(km_numFactors, 0.0)

// Populare factori la prima bară
if barstate.isfirst
    for i = 0 to km_numFactors - 1
        float _f = i_kmeansMinMult + i * i_kmeansStep
        array.set(km_factors, i, _f)

// ═══════════════════════════════════════════════════════════════════════════════
// CALCULARE SUPERTREND PENTRU FIECARE FACTOR
// ═══════════════════════════════════════════════════════════════════════════════

// Variabile pentru cel mai bun Supertrend
var float km_bestOutput = na
var int km_bestTrend = 0
var float km_bestPerf = 0.0
var float km_bestFactor = i_atrMult
var float km_bestUpper = na
var float km_bestLower = na

// Calculăm pentru fiecare factor (optimizat cu variabile separate)

// HYPER WAVE OSCILLATOR
// ═══════════════════════════════════════════════════════════════════════════════

f_hyperWave(int _length, int _signalLen) =>
    float _hi = ta.highest(high, _length)
    float _lo = ta.lowest(low, _length)
    float _av = ta.sma(hl2, _length)
    
    // Calculare oscillator
    float _range = _hi - _lo
    float _mid = math.avg(_hi, _lo, _av)
    float _raw = _range > 0 ? (close - _mid) / _range * 100 : 0
    
    // Smoothing cu EMA și linreg
    float _signal = ta.ema(ta.linreg(_raw, _length, 0), _signalLen)
    float _signalLine = ta.sma(_signal, 2)
    
    // Determinare culoare
    color _color = _signal > _signalLine ? C_BULL : C_BEAR
    
    // Crossovers
    bool _crossUp = ta.crossover(_signal, _signalLine)
    bool _crossDown = ta.crossunder(_signal, _signalLine)
    
    [_signal, _signalLine, _color, _crossUp, _crossDown]

// Aplicare Hyper Wave
[hyperWaveSignal, hyperWaveSignalLine, hyperWaveColor, hyperWaveCrossUp, hyperWaveCrossDown] = f_hyperWave(i_hyperWaveLen, i_hyperWaveSignal)

// SMART MONEY FLOW (MFI MODIFICAT)
// ═══════════════════════════════════════════════════════════════════════════════

f_smartMoneyFlow(int _length, int _smooth) =>
    // Money Flow Index standard
    float _mfiRaw = ta.mfi(hl2, _length)
    
    // Normalizare la -50 / +50
    float _mfi = ta.sma(_mfiRaw - 50, _smooth)
    
    // Culoare bazată pe direcție
    color _color = _mfi > 0 ? C_BULL : C_BEAR
    
    // Threshold-uri pentru bull/bear puternic
    float _bullThreshold = ta.sma(_mfi > 0 ? _mfi : 0, _length)
    float _bearThreshold = ta.sma(_mfi < 0 ? _mfi : 0, _length)
    
    // Semnale de divergență
    bool _bullSignal = _mfi > 0 and _mfi > _bullThreshold
    bool _bearSignal = _mfi < 0 and _mfi < _bearThreshold
    
    [_mfi, _color, _bullThreshold, _bearThreshold, _bullSignal, _bearSignal]

f_detectReversal(int _factor) =>
    float _volMA = ta.sma(volume, 7)
    float _rsi = ta.rsi(_volMA, 7) - 50
    
    // Major reversal: volum foarte mare
    float _majorThresh = _factor != 10 ? 1 + (_factor / 10.0) : 2.0
    bool _majorVol = volume > _volMA * _majorThresh
    
    // Minor reversal: volum moderat
    float _minorThresh = _factor != 10 ? 0 + (_factor / 10.0) : 2.0
    bool _minorVol = volume > _volMA * _minorThresh and not _majorVol
    
    // Condiții de reversal
    bool _majorBullRev = _majorVol and hyperWaveSignal < -20 and smfValue < smfBearThresh
    bool _majorBearRev = _majorVol and hyperWaveSignal > 20 and smfValue > smfBullThresh
    
    bool _minorBullRev = _minorVol and hyperWaveSignal < -10 and hyperWaveSignal > hyperWaveSignalLine and _rsi < -20
    bool _minorBearRev = _minorVol and hyperWaveSignal > 10 and hyperWaveSignal < hyperWaveSignalLine and _rsi > 20
    
    [_majorBullRev, _majorBearRev, _minorBullRev, _minorBearRev]

// Aplicare Reversal Detection

f_aiSignalClassifier(bool _isBuy) =>
    int _quality = 1
    float _score = 0.0
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 1: Trend Alignment (max 1.0)
    // ═══════════════════════════════════════════════════════════════════════════
    
    // EMA alignment
    bool _emaBull = g_ema8 > g_ema21 and g_ema21 > g_ema50
    bool _emaBear = g_ema8 < g_ema21 and g_ema21 < g_ema50
    
    if (_isBuy and _emaBull) or (not _isBuy and _emaBear)
        _score += 1.0
    else if (_isBuy and g_ema8 > g_ema21) or (not _isBuy and g_ema8 < g_ema21)
        _score += 0.5
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 2: Indicator Confluence (max 1.0)
    // ═══════════════════════════════════════════════════════════════════════════
    
    float _confPct = _isBuy ? confBullPct : confBearPct
    _score += _confPct / 100.0
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 3: Volume Confirmation (max 0.5)
    // ═══════════════════════════════════════════════════════════════════════════
    
    float _volRatio = f_volumeRatio(20)
    if _volRatio > 1.5
        _score += 0.5
    else if _volRatio > 1.0
        _score += 0.25
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 4: ADX Strength (max 0.5)
    // ═══════════════════════════════════════════════════════════════════════════
    
    if g_adx > 30
        _score += 0.5
    else if g_adx > 20
        _score += 0.3
    else if g_adx > 15
        _score += 0.1
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 5: WaveTrend Alignment (max 0.5)
    // ═══════════════════════════════════════════════════════════════════════════
    
    if (_isBuy and wtAllBullish) or (not _isBuy and wtAllBearish)
        _score += 0.5
    else if (_isBuy and wt3d_normal > 0) or (not _isBuy and wt3d_normal < 0)
        _score += 0.25
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 6: K-Means Performance (max 0.5)
    // ═══════════════════════════════════════════════════════════════════════════
    
    _score += kmeans_perfIndex * 0.5
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 7: Market Condition (max 0.5)
    // ═══════════════════════════════════════════════════════════════════════════
    
    // Piață trending = bonus
    if isTrending
        _score += 0.5
    else if not isChoppy
        _score += 0.25
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 8: Smart Money Alignment (max 0.5)
    // ═══════════════════════════════════════════════════════════════════════════
    
    if (_isBuy and smfBullSignal) or (not _isBuy and smfBearSignal)
        _score += 0.5
    else if (_isBuy and smfValue > 0) or (not _isBuy and smfValue < 0)
        _score += 0.2
    
    // ═══════════════════════════════════════════════════════════════════════════
    // FACTOR 9: Divergence Bonus (max 0.5)
    // ═══════════════════════════════════════════════════════════════════════════
    
    if (_isBuy and (oscBullDivergence or wtBullDiv)) or (not _isBuy and (oscBearDivergence or wtBearDiv))
        _score += 0.5
    
    // ═══════════════════════════════════════════════════════════════════════════
    // PENALIZĂRI
    // ═══════════════════════════════════════════════════════════════════════════
    
    // Penalizare pentru piață choppy
    if isChoppy
        _score -= 0.5
    
    // Penalizare pentru trend contrar
    if (_isBuy and _emaBear) or (not _isBuy and _emaBull)
        _score -= 0.3
    
    // Penalizare pentru volum scăzut
    if _volRatio < 0.5
        _score -= 0.2
    
    // ═══════════════════════════════════════════════════════════════════════════
    // CONVERSIE SCOR LA CALITATE 1-4
    // ═══════════════════════════════════════════════════════════════════════════
    
    _quality := _score >= 4.0 ? 4 : _score >= 3.0 ? 3 : _score >= 2.0 ? 2 : 1
    
    [_quality, _score]

// ═══════════════════════════════════════════════════════════════════════════════
// PARTEA 9/15: PATTERN DETECTION - FUNCȚII BAZĂ
// ═══════════════════════════════════════════════════════════════════════════════
// Funcții fundamentale pentru detectarea pattern-urilor
// Pivot points, trend lines, structură preț
// ═══════════════════════════════════════════════════════════════════════════════

// ═══════════════════════════════════════════════════════════════════════════════
// PARAMETRI ADAPTIVI PE TIMEFRAME
// ═══════════════════════════════════════════════════════════════════════════════

float g_tfMinutes = f_tfToMinutes()

// Lookback adaptat la TF
int g_patLookback = g_tfMinutes <= 1 ? 15 : 
     g_tfMinutes <= 5 ? 20 : 
     g_tfMinutes <= 15 ? 25 : 
     g_tfMinutes <= 60 ? 30 : 
     g_tfMinutes <= 240 ? 40 : 
     g_tfMinutes <= 1440 ? 50 : 60

// Toleranță adaptat la TF
float g_patTolerance = g_tfMinutes <= 5 ? 0.3 : 
     g_tfMinutes <= 15 ? 0.4 : 
     g_tfMinutes <= 60 ? 0.5 : 
     g_tfMinutes <= 240 ? 0.6 : 0.7

// Multiplicator impuls
float g_patImpulseMult = g_tfMinutes <= 1 ? 1.0 : 
     g_tfMinutes <= 5 ? 1.2 : 
     g_tfMinutes <= 15 ? 1.5 : 
     g_tfMinutes <= 60 ? 1.8 : 
     g_tfMinutes <= 240 ? 2.0 : 2.5

// Lungime consolidare
int g_patConsolLen = g_tfMinutes <= 5 ? 5 : 
     g_tfMinutes <= 15 ? 6 : 
     g_tfMinutes <= 60 ? 8 : 
     g_tfMinutes <= 240 ? 10 : 12

// Fiabilitate TF (TF mai mari = mai fiabile)
float g_tfReliability = g_tfMinutes <= 1 ? 0.4 : 
     g_tfMinutes <= 5 ? 0.55 : 
     g_tfMinutes <= 15 ? 0.65 : 
     g_tfMinutes <= 60 ? 0.75 : 
     g_tfMinutes <= 240 ? 0.85 : 
     g_tfMinutes <= 1440 ? 0.92 : 0.95

// Pivot lookback
int g_pivotLB = g_tfMinutes <= 5 ? 3 : 
     g_tfMinutes <= 15 ? 4 : 
     g_tfMinutes <= 60 ? 5 : 
     g_tfMinutes <= 240 ? 6 : 8

// ═══════════════════════════════════════════════════════════════════════════════
// STRUCTURĂ PREȚ - SWING HIGHS/LOWS
// ═══════════════════════════════════════════════════════════════════════════════

float g_swingHi = ta.highest(high, g_patConsolLen)
float g_swingLo = ta.lowest(low, g_patConsolLen)
float g_swingHiMed = ta.highest(high, math.min(i_patternLookback, 50))
float g_swingLoMed = ta.lowest(low, math.min(i_patternLookback, 50))

float g_consolRange = g_swingHi - g_swingLo
float g_totalRange = g_swingHiMed - g_swingLoMed
bool g_consolTight = g_totalRange > 0 ? (g_consolRange / g_totalRange) < 0.5 : false

// ═══════════════════════════════════════════════════════════════════════════════
// PIVOT POINTS DETECTION
// ═══════════════════════════════════════════════════════════════════════════════

float g_pivotHigh = ta.pivothigh(high, g_pivotLB, g_pivotLB)
float g_pivotLow = ta.pivotlow(low, g_pivotLB, g_pivotLB)

// Arrays pentru stocare pivot points
var array<float> g_pivotHighs = array.new<float>()
var array<float> g_pivotLows = array.new<float>()
var array<int> g_pivotHighBars = array.new<int>()
var array<int> g_pivotLowBars = array.new<int>()

// Actualizare pivot arrays
if not na(g_pivotHigh)
    if array.size(g_pivotHighs) >= 10
        array.shift(g_pivotHighs)
        array.shift(g_pivotHighBars)
    array.push(g_pivotHighs, g_pivotHigh)
    array.push(g_pivotHighBars, bar_index - g_pivotLB)

if not na(g_pivotLow)
    if array.size(g_pivotLows) >= 10
        array.shift(g_pivotLows)
        array.shift(g_pivotLowBars)
    array.push(g_pivotLows, g_pivotLow)
    array.push(g_pivotLowBars, bar_index - g_pivotLB)

// ═══════════════════════════════════════════════════════════════════════════════
// TREND LINE SLOPES
// ═══════════════════════════════════════════════════════════════════════════════

float g_sma10 = ta.sma(close, 10)
float g_sma20 = ta.sma(close, 20)
float g_slope10 = g_sma10 - g_sma10[math.min(g_patConsolLen, 5)]
float g_slope20 = g_sma20 - g_sma20[math.min(g_patConsolLen, 5)]

float g_upperSlope = (g_swingHiMed - g_swingHiMed[i_patternLookback > 1 ? i_patternLookback - 1 : 1]) / i_patternLookback
float g_lowerSlope = (g_swingLoMed - g_swingLoMed[i_patternLookback > 1 ? i_patternLookback - 1 : 1]) / i_patternLookback

// ═══════════════════════════════════════════════════════════════════════════════
// VOLUME ANALYSIS PENTRU PATTERNS
// ═══════════════════════════════════════════════════════════════════════════════

// Volume pentru patterns
float g_volAvgAdapt = ta.sma(volume, g_tfMinutes <= 15 ? 15 : 20)
bool g_volDecliningPat = volume < g_volAvgAdapt * 0.8
bool g_volIncreasingPatPat = volume > g_volAvgAdapt * 1.2

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Verifică dacă prețuri sunt la același nivel
// ═══════════════════════════════════════════════════════════════════════════════

f_pricesEqual(float _p1, float _p2, float _tolerance) =>
    float _diff = math.abs(_p1 - _p2)
    float _avg = (_p1 + _p2) / 2
    _avg > 0 ? (_diff / _avg) < _tolerance : false

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Calculează measured move pentru target
// ═══════════════════════════════════════════════════════════════════════════════

f_measuredMove(float _height, float _breakLevel, bool _isUp) =>
    _isUp ? _breakLevel + _height : _breakLevel - _height

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Detectare Higher Highs / Lower Lows
// ═══════════════════════════════════════════════════════════════════════════════

f_detectHHLL() =>
    bool _hh = false
    bool _hl = false
    bool _lh = false
    bool _ll = false
    
    if array.size(g_pivotHighs) >= 2
        float _lastHi = array.get(g_pivotHighs, array.size(g_pivotHighs) - 1)
        float _prevHi = array.get(g_pivotHighs, array.size(g_pivotHighs) - 2)
        _hh := _lastHi > _prevHi
        _lh := _lastHi < _prevHi
    
    if array.size(g_pivotLows) >= 2
        float _lastLo = array.get(g_pivotLows, array.size(g_pivotLows) - 1)
        float _prevLo = array.get(g_pivotLows, array.size(g_pivotLows) - 2)
        _hl := _lastLo > _prevLo
        _ll := _lastLo < _prevLo
    
    [_hh, _hl, _lh, _ll]

[pat_HH, pat_HL, pat_LH, pat_LL] = f_detectHHLL()

// Trend structure
bool g_uptrendStructure = pat_HH and pat_HL
bool g_downtrendStructure = pat_LH and pat_LL

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Detectare Impuls (mișcare puternică)
// ═══════════════════════════════════════════════════════════════════════════════

f_detectImpulse(int _lookback, bool _isUp) =>
    bool _found = false
    float _impulseSize = 0.0
    int _impulseBar = 0
    
    for i = g_patConsolLen to math.min(_lookback, 30)
        float _move = _isUp ? (close[i] - open[i]) : (open[i] - close[i])
        bool _isImpulse = _move > g_atr * g_patImpulseMult
        
        if _isImpulse and not _found
            _found := true
            _impulseSize := _move
            _impulseBar := i
            break
    
    [_found, _impulseSize, _impulseBar]

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Detectare Consolidare
// ═══════════════════════════════════════════════════════════════════════════════

f_detectConsolidation(int _length) =>
    float _rangeRecent = ta.highest(high, _length) - ta.lowest(low, _length)
    float _rangePrev = ta.highest(high[_length], _length) - ta.lowest(low[_length], _length)
    
    bool _isConsolidating = _rangePrev > 0 ? (_rangeRecent / _rangePrev) < 0.6 : false
    float _consolidationRatio = _rangePrev > 0 ? _rangeRecent / _rangePrev : 1.0
    
    [_isConsolidating, _consolidationRatio, _rangeRecent]

[g_isConsolidating, g_consolRatio, g_recentRange] = f_detectConsolidation(g_patConsolLen)

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Detectare Convergență/Divergență linii
// ═══════════════════════════════════════════════════════════════════════════════

f_detectConvergence() =>
    float _hiPrev = ta.highest(high[g_patConsolLen], g_patConsolLen)
    float _loPrev = ta.lowest(low[g_patConsolLen], g_patConsolLen)
    float _loRecent = ta.lowest(low, g_patConsolLen)
    
    bool _converging = g_swingHi < _hiPrev and _loRecent > _loPrev
    bool _diverging = g_swingHi > _hiPrev and _loRecent < _loPrev
    bool _parallelUp = g_swingHi > _hiPrev and _loRecent > _loPrev
    bool _parallelDown = g_swingHi < _hiPrev and _loRecent < _loPrev
    
    [_converging, _diverging, _parallelUp, _parallelDown, _hiPrev, _loPrev, _loRecent]

[g_converging, g_diverging, g_parallelUp, g_parallelDown, g_hiPrev, g_loPrev, g_loRecent] = f_detectConvergence()

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Detectare Breakout
// ═══════════════════════════════════════════════════════════════════════════════

f_detectBreakout() =>
    float _resistance = ta.highest(high[1], 20)
    float _support = ta.lowest(low[1], 20)
    
    bool _breakoutUp = close > _resistance and close[1] <= _resistance
    bool _breakoutDown = close < _support and close[1] >= _support
    
    bool _retestUp = high >= _resistance and close < _resistance and close > _support
    bool _retestDown = low <= _support and close > _support and close < _resistance
    
    [_breakoutUp, _breakoutDown, _retestUp, _retestDown, _resistance, _support]

[g_breakoutUp, g_breakoutDown, g_retestUp, g_retestDown, g_resistance, g_support] = f_detectBreakout()

// ═══════════════════════════════════════════════════════════════════════════════
// FUNCȚIE: Calculare Neckline pentru H&S patterns
// ═══════════════════════════════════════════════════════════════════════════════

f_calcNeckline() =>
    float _neckline = na
    
    if array.size(g_pivotLows) >= 2
        float _lo1 = array.get(g_pivotLows, array.size(g_pivotLows) - 1)
        float _lo2 = array.get(g_pivotLows, array.size(g_pivotLows) - 2)
        _neckline := math.avg(_lo1, _lo2)
    
    _neckline

float g_neckline = f_calcNeckline()

// 1. CLASSIC PATTERNS - Double Top/Bottom, Triple, Head & Shoulders

f_detectClassicPatterns() =>
    if not i_detectPatterns or not i_detectClassic
        false
    else
        bool _found = false
        
        // ═══════════════════════════════════════════════════════════════════════
        // DOUBLE TOP
        // ═══════════════════════════════════════════════════════════════════════
        if array.size(g_pivotHighs) >= 2 and not _found
            float _h1 = array.get(g_pivotHighs, array.size(g_pivotHighs) - 2)
            float _h2 = array.get(g_pivotHighs, array.size(g_pivotHighs) - 1)
            
            if f_pricesEqual(_h1, _h2, i_patternSensitivity * 2)
                float _neckline = g_swingLoMed
                if close < _neckline * 0.995
                    float _height = math.avg(_h1, _h2) - _neckline
                    float _target = _neckline - _height
                    f_setPattern("DOUBLE TOP", "bearish", 0.85, _target, 
                         "✓ Două vârfuri la același nivel\n✓ Neckline spart în jos",
                         "Pattern de reversare bearish.\nȚintă: " + str.tostring(_target, format.mintick),
                         "CLASSIC", _h1, _neckline)
                    _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // DOUBLE BOTTOM
        // ═══════════════════════════════════════════════════════════════════════
        if array.size(g_pivotLows) >= 2 and not _found
            float _l1 = array.get(g_pivotLows, array.size(g_pivotLows) - 2)
            float _l2 = array.get(g_pivotLows, array.size(g_pivotLows) - 1)
            
            if f_pricesEqual(_l1, _l2, i_patternSensitivity * 2)
                float _neckline = g_swingHiMed
                if close > _neckline * 1.005
                    float _height = _neckline - math.avg(_l1, _l2)
                    float _target = _neckline + _height
                    f_setPattern("DOUBLE BOTTOM", "bullish", 0.85, _target,
                         "✓ Două minime la același nivel\n✓ Neckline spart în sus",
                         "Pattern de reversare bullish.\nȚintă: " + str.tostring(_target, format.mintick),
                         "CLASSIC", _neckline, _l1)
                    _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // TRIPLE TOP
        // ═══════════════════════════════════════════════════════════════════════
        if array.size(g_pivotHighs) >= 3 and not _found
            float _h1 = array.get(g_pivotHighs, array.size(g_pivotHighs) - 3)
            float _h2 = array.get(g_pivotHighs, array.size(g_pivotHighs) - 2)
            float _h3 = array.get(g_pivotHighs, array.size(g_pivotHighs) - 1)
            
            if f_pricesEqual(_h1, _h2, i_patternSensitivity * 2) and f_pricesEqual(_h2, _h3, i_patternSensitivity * 2)
                float _neckline = g_swingLoMed
                float _height = math.avg(_h1, _h2, _h3) - _neckline
                float _target = _neckline - _height
                f_setPattern("TRIPLE TOP", "bearish", 0.90, _target,
                     "✓ Trei vârfuri la același nivel\n✓ Rezistență puternică confirmată",
                     "Pattern bearish foarte puternic.\nȚintă: " + str.tostring(_target, format.mintick),
                     "CLASSIC", _h1, _neckline)
                _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // TRIPLE BOTTOM
        // ═══════════════════════════════════════════════════════════════════════
        if array.size(g_pivotLows) >= 3 and not _found
            float _l1 = array.get(g_pivotLows, array.size(g_pivotLows) - 3)
            float _l2 = array.get(g_pivotLows, array.size(g_pivotLows) - 2)
            float _l3 = array.get(g_pivotLows, array.size(g_pivotLows) - 1)
            
            if f_pricesEqual(_l1, _l2, i_patternSensitivity * 2) and f_pricesEqual(_l2, _l3, i_patternSensitivity * 2)
                float _neckline = g_swingHiMed
                float _height = _neckline - math.avg(_l1, _l2, _l3)
                float _target = _neckline + _height
                f_setPattern("TRIPLE BOTTOM", "bullish", 0.90, _target,
                     "✓ Trei minime la același nivel\n✓ Suport puternic confirmat",
                     "Pattern bullish foarte puternic.\nȚintă: " + str.tostring(_target, format.mintick),
                     "CLASSIC", _neckline, _l1)
                _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // HEAD & SHOULDERS
        // ═══════════════════════════════════════════════════════════════════════
        if array.size(g_pivotHighs) >= 3 and not _found
            float _left = array.get(g_pivotHighs, array.size(g_pivotHighs) - 3)
            float _head = array.get(g_pivotHighs, array.size(g_pivotHighs) - 2)
            float _right = array.get(g_pivotHighs, array.size(g_pivotHighs) - 1)
            
            // Head trebuie să fie mai sus decât shoulders
            bool _isHS = _head > _left and _head > _right
            // Shoulders aproximativ egale
            bool _shouldersEqual = f_pricesEqual(_left, _right, i_patternSensitivity * 3)
            
            if _isHS and _shouldersEqual
                float _neckline = na(g_neckline) ? g_swingLoMed : g_neckline
                float _height = _head - _neckline
                float _target = _neckline - _height
                f_setPattern("HEAD & SHOULDERS", "bearish", 0.88, _target,
                     "✓ Cap mai înalt decât umerii\n✓ Umeri aproximativ egali\n✓ Neckline identificat",
                     "Pattern clasic de reversare.\nȚintă: " + str.tostring(_target, format.mintick),
                     "CLASSIC", _head, _neckline)
                _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // INVERSE HEAD & SHOULDERS
        // ═══════════════════════════════════════════════════════════════════════
        if array.size(g_pivotLows) >= 3 and not _found
            float _left = array.get(g_pivotLows, array.size(g_pivotLows) - 3)
            float _head = array.get(g_pivotLows, array.size(g_pivotLows) - 2)
            float _right = array.get(g_pivotLows, array.size(g_pivotLows) - 1)
            
            bool _isIHS = _head < _left and _head < _right
            bool _shouldersEqual = f_pricesEqual(_left, _right, i_patternSensitivity * 3)
            
            if _isIHS and _shouldersEqual
                float _neckline = g_swingHiMed
                float _height = _neckline - _head
                float _target = _neckline + _height
                f_setPattern("INVERSE H&S", "bullish", 0.88, _target,
                     "✓ Cap mai jos decât umerii\n✓ Umeri aproximativ egali\n✓ Pattern de acumulare",
                     "Pattern bullish puternic.\nȚintă: " + str.tostring(_target, format.mintick),
                     "CLASSIC", _neckline, _head)
                _found := true
        
        _found

// 2. TRIANGLE PATTERNS
// ═══════════════════════════════════════════════════════════════════════════════

f_detectTrianglePatterns() =>
    if not i_detectPatterns or not i_detectTriangles
        false
    else
        bool _found = false
        
        // ═══════════════════════════════════════════════════════════════════════
        // ASCENDING TRIANGLE
        // ═══════════════════════════════════════════════════════════════════════
        bool _flatResistance = math.abs(g_swingHi - g_hiPrev) < g_atr * g_patTolerance
        bool _risingSupport = g_loRecent > g_loPrev
        
        if _flatResistance and _risingSupport and g_volDecliningPat and g_consolTight and not _found
            float _height = g_swingHi - g_swingLo
            float _target = g_swingHi + _height
            f_setPattern("ASCENDING TRIANGLE", "bullish", 0.78, _target,
                 "✓ Rezistență plată\n✓ Suport în creștere\n✓ Volum în scădere",
                 "Breakout probabil în sus.\nȚintă: " + str.tostring(_target, format.mintick),
                 "TRIANGLE", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // DESCENDING TRIANGLE
        // ═══════════════════════════════════════════════════════════════════════
        bool _flatSupport = math.abs(g_loRecent - g_loPrev) < g_atr * g_patTolerance
        bool _fallingResistance = g_swingHi < g_hiPrev
        
        if _flatSupport and _fallingResistance and g_volDecliningPat and g_consolTight and not _found
            float _height = g_swingHi - g_swingLo
            float _target = g_swingLo - _height
            f_setPattern("DESCENDING TRIANGLE", "bearish", 0.78, _target,
                 "✓ Suport plat\n✓ Rezistență în scădere\n✓ Volum în scădere",
                 "Breakout probabil în jos.\nȚintă: " + str.tostring(_target, format.mintick),
                 "TRIANGLE", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // SYMMETRICAL TRIANGLE
        // ═══════════════════════════════════════════════════════════════════════
        if g_converging and g_volDecliningPat and g_consolTight and not _found
            float _height = g_totalRange * 0.75
            string _dir = close > g_sma20 ? "bullish" : "bearish"
            float _target = _dir == "bullish" ? close + _height : close - _height
            f_setPattern("SYMMETRICAL TRIANGLE", _dir, 0.72, _target,
                 "✓ Linii convergente\n✓ Volum în scădere\n✓ Așteptare breakout",
                 "Direcția breakout determină trendul.\nȚintă: " + str.tostring(_target, format.mintick),
                 "TRIANGLE", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // EXPANDING TRIANGLE (Megaphone)
        // ═══════════════════════════════════════════════════════════════════════
        if g_diverging and g_volIncreasingPat and not _found
            f_setPattern("EXPANDING TRIANGLE", "neutral", 0.65, na,
                 "✓ Linii divergente\n✓ Volatilitate în creștere\n✓ Incertitudine mare",
                 "Pattern de volatilitate.\nAtenție la false breakouts!",
                 "TRIANGLE", g_swingHi, g_swingLo)
            _found := true
        
        _found

// 3. CONTINUATION PATTERNS
// ═══════════════════════════════════════════════════════════════════════════════

f_detectContinuationPatterns() =>
    if not i_detectPatterns or not i_detectContinuation
        false
    else
        bool _found = false
        
        // Detectare impuls anterior
        [_hadImpulseUp, _impulseUpSize, _impulseUpBar] = f_detectImpulse(i_patternLookback, true)
        [_hadImpulseDown, _impulseDownSize, _impulseDownBar] = f_detectImpulse(i_patternLookback, false)
        
        // ═══════════════════════════════════════════════════════════════════════
        // BULL FLAG
        // ═══════════════════════════════════════════════════════════════════════
        bool _mildDownSlope = g_slope10 < 0 and g_slope10 > -(g_atr * 0.3)
        
        if _hadImpulseUp and g_consolTight and _mildDownSlope and g_volDecliningPat and not _found
            float _target = close + _impulseUpSize * 0.8
            f_setPattern("BULL FLAG", "bullish", 0.80, _target,
                 "✓ Impuls puternic în sus\n✓ Consolidare cu pantă ușoară în jos\n✓ Volum în scădere",
                 "Continuare bullish probabilă.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CONTINUATION", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // BEAR FLAG
        // ═══════════════════════════════════════════════════════════════════════
        bool _mildUpSlope = g_slope10 > 0 and g_slope10 < (g_atr * 0.3)
        
        if _hadImpulseDown and g_consolTight and _mildUpSlope and g_volDecliningPat and not _found
            float _target = close - _impulseDownSize * 0.8
            f_setPattern("BEAR FLAG", "bearish", 0.80, _target,
                 "✓ Impuls puternic în jos\n✓ Consolidare cu pantă ușoară în sus\n✓ Volum în scădere",
                 "Continuare bearish probabilă.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CONTINUATION", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // BULL PENNANT
        // ═══════════════════════════════════════════════════════════════════════
        bool _tightConverging = g_converging and g_recentRange < g_atr * 1.5
        
        if _hadImpulseUp and _tightConverging and g_volDecliningPat and not _found
            float _target = close + _impulseUpSize * 0.7
            f_setPattern("BULL PENNANT", "bullish", 0.75, _target,
                 "✓ Impuls în sus (flagpole)\n✓ Consolidare în triunghi\n✓ Pregătire breakout",
                 "Similar cu flag, dar simetric.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CONTINUATION", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // BEAR PENNANT
        // ═══════════════════════════════════════════════════════════════════════
        if _hadImpulseDown and _tightConverging and g_volDecliningPat and not _found
            float _target = close - _impulseDownSize * 0.7
            f_setPattern("BEAR PENNANT", "bearish", 0.75, _target,
                 "✓ Impuls în jos (flagpole)\n✓ Consolidare în triunghi\n✓ Pregătire breakdown",
                 "Continuare bearish.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CONTINUATION", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // RISING WEDGE (bearish)
        // ═══════════════════════════════════════════════════════════════════════
        bool _bothUp = g_upperSlope > 0 and g_lowerSlope > 0
        bool _narrowing = g_converging
        bool _upperSteeper = g_upperSlope < g_lowerSlope // Lower converge mai repede
        
        if _bothUp and _narrowing and g_volDecliningPat and not _found
            float _target = close - g_totalRange * 0.618
            f_setPattern("RISING WEDGE", "bearish", 0.76, _target,
                 "✓ Ambele linii în creștere\n✓ Convergență\n✓ Slăbire momentum",
                 "Pattern bearish de reversare.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CONTINUATION", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // FALLING WEDGE (bullish)
        // ═══════════════════════════════════════════════════════════════════════
        bool _bothDown = g_upperSlope < 0 and g_lowerSlope < 0
        
        if _bothDown and _narrowing and g_volDecliningPat and not _found
            float _target = close + g_totalRange * 0.618
            f_setPattern("FALLING WEDGE", "bullish", 0.76, _target,
                 "✓ Ambele linii în scădere\n✓ Convergență\n✓ Acumulare",
                 "Pattern bullish de reversare.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CONTINUATION", g_swingHi, g_swingLo)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // RECTANGLE / RANGE
        // ═══════════════════════════════════════════════════════════════════════
        bool _flatRange = math.abs(g_upperSlope) < g_atr * 0.1 and math.abs(g_lowerSlope) < g_atr * 0.1
        
        if _flatRange and g_isConsolidating and not _found
            string _dir = close > g_sma20 ? "bullish" : "bearish"
            float _target = _dir == "bullish" ? g_swingHi + g_consolRange : g_swingLo - g_consolRange
            f_setPattern("RECTANGLE", _dir, 0.70, _target,
                 "✓ Range plat\n✓ Suport și rezistență clare\n✓ Așteptare breakout",
                 "Breakout în direcția trendului.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CONTINUATION", g_swingHi, g_swingLo)
            _found := true
        
        _found

// 4. PRICE ACTION PATTERNS (Candlestick)
// ═══════════════════════════════════════════════════════════════════════════════

f_detectPriceActionPatterns() =>
    if not i_detectPatterns or not i_detectPriceAction
        false
    else
        bool _found = false
        
        // Componente candle
        float _body = math.abs(close - open)
        float _upperWick = high - math.max(close, open)
        float _lowerWick = math.min(close, open) - low
        float _fullRange = high - low
        float _avgBody = ta.sma(math.abs(close - open), 20)
        
        // ═══════════════════════════════════════════════════════════════════════
        // BULLISH PIN BAR / HAMMER
        // ═══════════════════════════════════════════════════════════════════════
        bool _bullPinBar = _lowerWick > _body * 2 and _upperWick < _body * 0.5 and _body > 0
        
        if _bullPinBar and not _found
            float _target = close + g_atr * 2
            f_setPattern("BULLISH PIN BAR", "bullish", 0.72, _target,
                 "✓ Fitil inferior > 2x corp\n✓ Fitil superior mic\n✓ Respingere de la minim",
                 "Semnal de reversare bullish.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CANDLESTICK", high, low)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // BEARISH PIN BAR / SHOOTING STAR
        // ═══════════════════════════════════════════════════════════════════════
        bool _bearPinBar = _upperWick > _body * 2 and _lowerWick < _body * 0.5 and _body > 0
        
        if _bearPinBar and not _found
            float _target = close - g_atr * 2
            f_setPattern("BEARISH PIN BAR", "bearish", 0.72, _target,
                 "✓ Fitil superior > 2x corp\n✓ Fitil inferior mic\n✓ Respingere de la maxim",
                 "Semnal de reversare bearish.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CANDLESTICK", high, low)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // BULLISH ENGULFING
        // ═══════════════════════════════════════════════════════════════════════
        float _prevBody = math.abs(close[1] - open[1])
        bool _bullEngulf = close > open and close[1] < open[1] and close > open[1] and open < close[1] and _body > _prevBody * 1.2
        
        if _bullEngulf and not _found
            float _target = close + g_atr * 2.5
            f_setPattern("BULLISH ENGULFING", "bullish", 0.75, _target,
                 "✓ Corp bullish înghite corp bearish\n✓ Close > Open anterior\n✓ Semnal puternic",
                 "Reversare bullish confirmată.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CANDLESTICK", high, low)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // BEARISH ENGULFING
        // ═══════════════════════════════════════════════════════════════════════
        bool _bearEngulf = close < open and close[1] > open[1] and close < open[1] and open > close[1] and _body > _prevBody * 1.2
        
        if _bearEngulf and not _found
            float _target = close - g_atr * 2.5
            f_setPattern("BEARISH ENGULFING", "bearish", 0.75, _target,
                 "✓ Corp bearish înghite corp bullish\n✓ Close < Open anterior\n✓ Semnal puternic",
                 "Reversare bearish confirmată.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CANDLESTICK", high, low)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // DOJI
        // ═══════════════════════════════════════════════════════════════════════
        bool _isDoji = _body < _avgBody * 0.1 and _fullRange > _avgBody * 0.5
        
        if _isDoji and not _found
            string _dir = close > g_sma20 ? "bearish" : "bullish" // Doji în trend = potențial reversare
            f_setPattern("DOJI", _dir, 0.60, na,
                 "✓ Corp foarte mic\n✓ Indecizie în piață\n✓ Potențial punct de cotitură",
                 "Atenție la următoarea lumânare pentru confirmare.",
                 "CANDLESTICK", high, low)
            _found := true
        
        // ═══════════════════════════════════════════════════════════════════════
        // MORNING STAR (3 candles)
        // ═══════════════════════════════════════════════════════════════════════
        float _body2 = math.abs(close[1] - open[1])
        float _body3 = math.abs(close[2] - open[2])
        bool _morningStar = close[2] < open[2] and _body3 > _avgBody and 
             _body2 < _avgBody * 0.5 and 
             close > open and close > (open[2] + close[2]) / 2
        
        if _morningStar and not _found
            float _target = close + g_atr * 2.5
            f_setPattern("MORNING STAR", "bullish", 0.78, _target,
                 "✓ Lumânare bearish mare\n✓ Lumânare mică (indecizie)\n✓ Lumânare bullish de confirmare",
                 "Pattern puternic de reversare.\nȚintă: " + str.tostring(_target, format.mintick),
                 "CANDLESTICK", high, low[2])
            _found := true

f_detectHarmonicPatterns() =>
    if not i_detectPatterns or not i_detectHarmonic
        false
    else
        bool _found = false
        
        // Avem nevoie de cel puțin 4 pivot points
        if array.size(g_pivotHighs) >= 2 and array.size(g_pivotLows) >= 2
            // Obținem ultimele puncte
            float _X = array.size(g_pivotLows) >= 2 ? array.get(g_pivotLows, array.size(g_pivotLows) - 2) : na
            float _A = array.size(g_pivotHighs) >= 2 ? array.get(g_pivotHighs, array.size(g_pivotHighs) - 2) : na
            float _B = array.size(g_pivotLows) >= 1 ? array.get(g_pivotLows, array.size(g_pivotLows) - 1) : na
            float _C = array.size(g_pivotHighs) >= 1 ? array.get(g_pivotHighs, array.size(g_pivotHighs) - 1) : na
            float _D = close
            
            if not na(_X) and not na(_A) and not na(_B) and not na(_C)
                float _XA = _A - _X
                float _AB = _A - _B
                float _BC = _C - _B
                float _CD = _C - _D
                
                // ═══════════════════════════════════════════════════════════════
                // GARTLEY (AB = 0.618 XA, BC = 0.382-0.886 AB)
                // ═══════════════════════════════════════════════════════════════
                if f_checkFibRatio(_XA, _AB, 0.618, 0.08) and not _found
                    float _target = _D + _XA * 0.786
                    f_setPattern("GARTLEY", "bullish", 0.75, _target,
                         "✓ AB = 61.8% XA\n✓ Structură harmonică validă\n✓ D în zona PRZ",
                         "Pattern harmonic.\nȚintă: " + str.tostring(_target, format.mintick),
                         "HARMONIC", _C, _D)
                    _found := true
                
                // ═══════════════════════════════════════════════════════════════
                // BUTTERFLY (AB = 0.786 XA)
                // ═══════════════════════════════════════════════════════════════
                if f_checkFibRatio(_XA, _AB, 0.786, 0.08) and not _found
                    float _target = _D + _XA * 1.27
                    f_setPattern("BUTTERFLY", "bullish", 0.72, _target,
                         "✓ AB = 78.6% XA\n✓ Pattern extins\n✓ Reversare la D",
                         "Butterfly pattern.\nȚintă: " + str.tostring(_target, format.mintick),
                         "HARMONIC", _C, _D)
                    _found := true
                
                // ═══════════════════════════════════════════════════════════════
                // BAT (AB = 0.382-0.5 XA)
                // ═══════════════════════════════════════════════════════════════
                bool _batRatio = f_checkFibRatio(_XA, _AB, 0.382, 0.08) or f_checkFibRatio(_XA, _AB, 0.5, 0.08)
                if _batRatio and not _found
                    float _target = _D + _XA * 0.886
                    f_setPattern("BAT", "bullish", 0.70, _target,
                         "✓ AB = 38.2%-50% XA\n✓ Retracement profund\n✓ Zona de cumpărare",
                         "Bat pattern.\nȚintă: " + str.tostring(_target, format.mintick),
                         "HARMONIC", _C, _D)
                    _found := true
                
                // ═══════════════════════════════════════════════════════════════
                // CRAB (AB = 0.382-0.618 XA, CD extends beyond X)
                // ═══════════════════════════════════════════════════════════════
                if f_checkFibRatio(_XA, _AB, 0.618, 0.1) and _D < _X and not _found
                    float _target = _D + _XA * 1.618
                    f_setPattern("CRAB", "bullish", 0.68, _target,
                         "✓ Extensie extremă la D\n✓ D sub X\n✓ Potențial bounce mare",
                         "Deep crab pattern.\nȚintă: " + str.tostring(_target, format.mintick),
                         "HARMONIC", _C, _D)
                    _found := true
        
        _found

        float _recentHigh = ta.highest(high, 10)

// FUNCȚIE: Detectare pattern simplificat pe MTF
// ═══════════════════════════════════════════════════════════════════════════════
f_getMTFPattern(_symbol, _tf) =>
    // Obținem date
    float _close = request.security(_symbol, _tf, close[1], lookahead=barmerge.lookahead_off)
    float _open = request.security(_symbol, _tf, open[1], lookahead=barmerge.lookahead_off)
    float _high = request.security(_symbol, _tf, high[1], lookahead=barmerge.lookahead_off)
    float _low = request.security(_symbol, _tf, low[1], lookahead=barmerge.lookahead_off)
    float _close2 = request.security(_symbol, _tf, close[2], lookahead=barmerge.lookahead_off)
    float _open2 = request.security(_symbol, _tf, open[2], lookahead=barmerge.lookahead_off)
    
    float _highest = request.security(_symbol, _tf, ta.highest(high, 20)[1], lookahead=barmerge.lookahead_off)
    float _lowest = request.security(_symbol, _tf, ta.lowest(low, 20)[1], lookahead=barmerge.lookahead_off)
    
    // Pattern detection simplificat
    string _pattern = "—"
    float _prob = 0.0
    
    // Double Top/Bottom check
    float _h1 = request.security(_symbol, _tf, ta.highest(high, 10)[11], lookahead=barmerge.lookahead_off)
    float _h2 = request.security(_symbol, _tf, ta.highest(high, 10)[1], lookahead=barmerge.lookahead_off)
    float _l1 = request.security(_symbol, _tf, ta.lowest(low, 10)[11], lookahead=barmerge.lookahead_off)
    float _l2 = request.security(_symbol, _tf, ta.lowest(low, 10)[1], lookahead=barmerge.lookahead_off)
    
    // Engulfing
    float _body = math.abs(_close - _open)
    float _prevBody = math.abs(_close2 - _open2)
    bool _bullEngulf = _close > _open and _close2 < _open2 and _body > _prevBody * 1.2
    bool _bearEngulf = _close < _open and _close2 > _open2 and _body > _prevBody * 1.2
    
    // Pin bar
    float _upperWick = _high - math.max(_close, _open)
    float _lowerWick = math.min(_close, _open) - _low
    bool _bullPin = _lowerWick > _body * 2 and _upperWick < _body * 0.5
    bool _bearPin = _upperWick > _body * 2 and _lowerWick < _body * 0.5
    
    // Triangle check (simplified)
    float _range1 = request.security(_symbol, _tf, (ta.highest(high, 10) - ta.lowest(low, 10))[1], lookahead=barmerge.lookahead_off)
    float _range2 = request.security(_symbol, _tf, (ta.highest(high, 10) - ta.lowest(low, 10))[11], lookahead=barmerge.lookahead_off)
    bool _converging = _range1 < _range2 * 0.7
    
    // Determinare pattern
    if _bullEngulf
        _pattern := "Bull Engulf"
        _prob := 0.72
    else if _bearEngulf
        _pattern := "Bear Engulf"
        _prob := 0.72
    else if _bullPin
        _pattern := "Bull Pin"
        _prob := 0.68
    else if _bearPin
        _pattern := "Bear Pin"
        _prob := 0.68
    else if _converging
        _pattern := "Triangle"
        _prob := 0.65
    else if math.abs(_h1 - _h2) / _h1 < 0.01
        _pattern := "Double Top"
        _prob := 0.75
    else if math.abs(_l1 - _l2) / _l1 < 0.01
        _pattern := "Double Bot"
        _prob := 0.75
    
    [_pattern, _prob]

bool kmBuySignalRaw = kmeans_buy and trendStable
bool kmSellSignalRaw = kmeans_sell and trendStable

// ═══════════════════════════════════════════════════════════════════════════════
// CONDIȚII STRONG SIGNAL
// ═══════════════════════════════════════════════════════════════════════════════
bool strongBuyConditions = g_adx > i_adxThreshold and 
     (close > g_maTrend or close > kmeans_trailStop) and
     probFinal >= 0.65 and
     (trendCatcherDir == 1 or smartTrailTrend == 1)

bool strongSellConditions = g_adx > i_adxThreshold and 
     (close < g_maTrend or close < kmeans_trailStop) and
     probFinal <= 0.35 and
     (trendCatcherDir == -1 or smartTrailTrend == -1)
 
// ═══════════════════════════════════════════════════════════════════════════════
// SEMNALE FINALE
// ═══════════════════════════════════════════════════════════════════════════════

// Buy Signals
bool initialBuy = kmBuySignalRaw and probFilterPass and choppyFilterPass
bool strongBuy = initialBuy and strongBuyConditions

// Sell Signals
bool initialSell = kmSellSignalRaw and probFilterPass and choppyFilterPass
bool strongSell = initialSell and strongSellConditions

// Late Strong Signals (confirmare ulterioară)
var bool hasStrongFired = false
if kmBuySignalRaw or kmSellSignalRaw
    hasStrongFired := false

bool lateStrongBuy = kmeans_direction == 1 and not hasStrongFired and strongBuyConditions and not kmBuySignalRaw and trendStable and probFilterPass
bool lateStrongSell = kmeans_direction == -1 and not hasStrongFired and strongSellConditions and not kmSellSignalRaw and trendStable and probFilterPass

if (initialBuy and strongBuyConditions) or (initialSell and strongSellConditions) or lateStrongBuy or lateStrongSell
    hasStrongFired := true

// Actualizare variabile globale
g_buySignal := initialBuy or lateStrongBuy
g_sellSignal := initialSell or lateStrongSell
g_strongBuy := strongBuy or lateStrongBuy
g_strongSell := strongSell or lateStrongSell

// ═══════════════════════════════════════════════════════════════════════════════
// AI SIGNAL CLASSIFIER (1-4)
// ═══════════════════════════════════════════════════════════════════════════════

// Calculare calitate semnal
int signalQuality = 1

if g_buySignal or g_sellSignal
    bool isBuy = g_buySignal
    [quality, score] = f_aiSignalClassifier(isBuy)
    signalQuality := quality
    
    // Ajustare bazată pe WaveTrend quality
    signalQuality := math.max(signalQuality, wtSignalQuality)
    
    // Ajustare bazată pe K-Means quality
    signalQuality := math.max(signalQuality, kmeans_quality)
    
    // Cap la 4
    signalQuality := math.min(4, signalQuality)

g_signalQuality := signalQuality

// ═══════════════════════════════════════════════════════════════════════════════
// SISTEM TP/SL DINAMIC
// ═══════════════════════════════════════════════════════════════════════════════

// Ajustare multiplicatori bazat pe volatilitate
float volAdjust = g_volExpanding ? 1.3 : g_volCompressing ? 0.8 : 1.0
float probAdjust = probFinal >= 0.7 ? 1.2 : probFinal >= 0.5 ? 1.0 : 0.85

float tp1Mult = i_tpMultiplier1 * volAdjust * probAdjust
float tp2Mult = i_tpMultiplier2 * volAdjust * probAdjust
float tp3Mult = i_tpMultiplier3 * volAdjust * probAdjust
float slMult = i_slMultiplier * (g_volExpanding ? 1.3 : 1.0)

// ═══════════════════════════════════════════════════════════════════════════════
// SETARE NIVELURI TP/SL LA SEMNAL
// ═══════════════════════════════════════════════════════════════════════════════

if (initialBuy or lateStrongBuy) and probFilterPass
    // Ștergem vizualizările vechi
    line.delete(g_tp1Line)
    line.delete(g_tp2Line)
    line.delete(g_tp3Line)
    line.delete(g_slLine)
    label.delete(g_tp1Label)
    label.delete(g_tp2Label)
    label.delete(g_tp3Label)
    label.delete(g_slLabel)
    
    // Setăm niveluri noi
    g_trade.isActive := true
    g_trade.direction := 1
    g_trade.entryPrice := close
    g_trade.entryBar := bar_index
    g_trade.tp1 := close + g_atr * tp1Mult
    g_trade.tp2 := close + g_atr * tp2Mult
    g_trade.tp3 := close + g_atr * tp3Mult
    g_trade.sl := close - g_atr * slMult
    g_trade.tp1Hit := false
    g_trade.tp2Hit := false
    g_trade.tp3Hit := false
    g_trade.slHit := false

if (initialSell or lateStrongSell) and probFilterPass
    line.delete(g_tp1Line)
    line.delete(g_tp2Line)
    line.delete(g_tp3Line)
    line.delete(g_slLine)
    label.delete(g_tp1Label)
    label.delete(g_tp2Label)
    label.delete(g_tp3Label)
    label.delete(g_slLabel)

    g_trade.direction := -1
    g_trade.entryPrice := close
    g_trade.entryBar := bar_index
    g_trade.tp1 := close - g_atr * tp1Mult
    g_trade.tp2 := close - g_atr * tp2Mult
    g_trade.tp3 := close - g_atr * tp3Mult
    g_trade.sl := close + g_atr * slMult
    g_trade.tp1Hit := false
    g_trade.tp2Hit := false
    g_trade.tp3Hit := false
    g_trade.slHit := false

// TRAILING STOP LOSS
// ═══════════════════════════════════════════════════════════════════════════════
if i_trailingSL and g_trade.isActive
    if g_trade.direction == 1
        float newSL = close - g_atr * slMult
        if newSL > g_trade.sl
            g_trade.sl := newSL
    else
        float newSL = close + g_atr * slMult
        if newSL < g_trade.sl
            g_trade.sl := newSL

// VERIFICARE TP/SL ATINSE
// ═══════════════════════════════════════════════════════════════════════════════
if g_trade.isActive
    if g_trade.direction == 1
        if high >= g_trade.tp1 and not g_trade.tp1Hit
            g_trade.tp1Hit := true
        if high >= g_trade.tp2 and not g_trade.tp2Hit
            g_trade.tp2Hit := true
        if high >= g_trade.tp3 and not g_trade.tp3Hit
            g_trade.tp3Hit := true
        if low <= g_trade.sl
            g_trade.slHit := true
            g_trade.isActive := false
    else
        if low <= g_trade.tp1 and not g_trade.tp1Hit
            g_trade.tp1Hit := true
        if low <= g_trade.tp2 and not g_trade.tp2Hit
            g_trade.tp2Hit := true
        if low <= g_trade.tp3 and not g_trade.tp3Hit
            g_trade.tp3Hit := true
        if high >= g_trade.sl
            g_trade.slHit := true
            g_trade.isActive := false

// Invalidare trade la schimbare direcție

    if (g_trade.direction == 1 and close < kmeans_trailStop) or (g_trade.direction == -1 and close > kmeans_trailStop)
        g_trade.isActive := false

// CALCULARE PROBABILITĂȚI TP
// ═══════════════════════════════════════════════════════════════════════════════

f_calcTPProb(_tpLevel, _baseMult) =>
    if na(_tpLevel) or not g_trade.isActive
        0.0
    else
        float dist = math.abs(_tpLevel - close)
        float distATR = g_atr > 0 ? dist / g_atr : 5.0
        float decay = distATR <= 1.0 ? 1.0 : distATR <= 2.0 ? 0.85 : distATR <= 3.0 ? 0.65 : distATR <= 4.0 ? 0.45 : 0.25
        f_clamp(probFinal * _baseMult * decay, 0, 0.99) * 100

float probTP1 = f_calcTPProb(g_trade.tp1, 1.0)
float probTP2 = f_calcTPProb(g_trade.tp2, 0.75)
float probTP3 = f_calcTPProb(g_trade.tp3, 0.5)
